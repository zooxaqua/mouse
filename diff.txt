#pragma once

#include <stdint.h>

// =========================================================
// Controller Button Map
// =========================================================
struct PlatformControllerButtonMap
{
    uint32_t A;
    uint32_t B;
    uint32_t X;
    uint32_t Y;

    uint32_t L1;
    uint32_t R1;
    uint32_t L2;
    uint32_t R2;
    uint32_t L3;
    uint32_t R3;
};

// =========================================================
// Controller D-Pad Map
// =========================================================
struct PlatformControllerDpadMap
{
    uint8_t UP;
    uint8_t DOWN;
    uint8_t LEFT;
    uint8_t RIGHT;
};

// =========================================================
// Controller Misc Button Map
// =========================================================
struct PlatformControllerMiscButtonMap
{
    uint8_t HOME;
    uint8_t MINUS;
    uint8_t PLUS;
    uint8_t PICT;
};

// =========================================================
// Switch Controller Map
// =========================================================
constexpr PlatformControllerButtonMap g_switchButtonMap =
{
    .A  = 0x0002,
    .B  = 0x0001,
    .X  = 0x0008,
    .Y  = 0x0004,
    .L1 = 0x0010,
    .R1 = 0x0020,
    .L2 = 0x0040,
    .R2 = 0x0080,
    .L3 = 0x0100,
    .R3 = 0x0200
};

constexpr PlatformControllerDpadMap g_switchDpadMap =
{
    .UP    = 0x01,
    .DOWN  = 0x02,
    .LEFT  = 0x08,
    .RIGHT = 0x04
};

constexpr PlatformControllerMiscButtonMap g_switchMiscButtonMap =
{
    .HOME  = 0x01,
    .MINUS = 0x02,
    .PLUS  = 0x04,
    .PICT  = 0x08
};

// =========================================================
// Current Controller Map
// =========================================================
extern const PlatformControllerButtonMap* g_platformControllerButtonMap;
extern const PlatformControllerDpadMap* g_platformControllerDpadMap;
extern const PlatformControllerMiscButtonMap* g_platformControllerMiscButtonMap;


// =========================================================
// Controller Input
// =========================================================
struct PlatformControllerData
{
    int8_t leftStickX;
    int8_t leftStickY;
    int8_t rightStickX;
    int8_t rightStickY;

    uint32_t buttons;
    uint32_t buttonsPressed;
    uint32_t buttonsReleased;

    uint8_t dpad;
    uint8_t dpadPressed;
    uint8_t dpadReleased;

    uint8_t miscButtons;
    uint8_t miscButtonsPressed;
    uint8_t miscButtonsReleased;
};
